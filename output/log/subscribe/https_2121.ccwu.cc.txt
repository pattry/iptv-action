<script>

async function searchData(){

const keyword =
document
.getElementById("keyword")
.value
.trim();

if(!keyword){

alert("请输入频道名称");

return;

}

const textarea =
document.getElementById(
"resultText"
);

const stats =
document.getElementById(
"stats"
);

textarea.value =
"搜索中 请稍后...公众号【潇雨萌萌】";

stats.innerHTML = "";

try{

const res =
await fetch(
"/api/search?q=" +
encodeURIComponent(keyword)
);

const data =
await res.json();

if(data.error){

textarea.value =
"搜索失败";

return;

}

if(data.length === 0){

textarea.value =
"没有搜索到相关";

return;

}

stats.innerHTML =
"共搜索到 " +
data.length +
" 条结果";

textarea.value =
data.join("\n");

}catch(e){

textarea.value =
"搜索失败";

}

}

// 一键复制
function copyText(){

const textarea =
document.getElementById(
"resultText"
);

const text =
textarea.value;

if(!text){

alert("没有可复制内容");

return;

}

textarea.removeAttribute(
"readonly"
);

textarea.select();

textarea.setSelectionRange(
0,
999999
);

function fallback(){

try{

document.execCommand(
"copy"
);

alert("复制成功");

}catch(e){

alert(
"请手动复制"
);

}

}

if(
navigator.clipboard &&
window.isSecureContext
){

navigator.clipboard
.writeText(text)
.then(()=>{
alert("复制成功");
})
.catch(()=>{
fallback();
});

}else{

fallback();

}

textarea.setAttribute(
"readonly",
true
);

}

</script>
